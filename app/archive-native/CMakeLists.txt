cmake_minimum_required(VERSION 3.18)
project(essentia_archive C)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(feature OPENSSL MD LIBB2 LZ4 LZO LZMA ZSTD BZip2 LIBXML2 BSDXML EXPAT PCREPOSIX PCRE2POSIX TAR CPIO CAT UNZIP XATTR ACL ICONV TEST INSTALL)
  set(ENABLE_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
include(FetchContent)
FetchContent_Declare(libarchive
  GIT_REPOSITORY https://github.com/libarchive/libarchive.git
  GIT_TAG 0c48412be2d94afe1859a807ea7896a18d8900bd)
FetchContent_MakeAvailable(libarchive)
add_library(essentia-archive SHARED archive_jni.c)
target_include_directories(essentia-archive PRIVATE ${libarchive_SOURCE_DIR}/libarchive)
if(NOT ANDROID)
  find_package(JNI REQUIRED)
  target_include_directories(essentia-archive PRIVATE ${JNI_INCLUDE_DIRS})
endif()
target_link_libraries(essentia-archive PRIVATE archive_static)
